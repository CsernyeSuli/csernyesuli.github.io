/* ===================== KÓDOK ===================== */

/** 31 karakter, a könnyen összetéveszthetők (0 O 1 I L) nélkül. A kód 7 véletlen karakter + 1 ellenőrző karakter. */
const CODE_ALPHABET = '23456789ABCDEFGHJKMNPQRSTUVWXYZ';
const CODE_BODY_LEN = 7;

/** Ellenőrző karakter: a súlyozott összeg mod 31 (egy elgépelt karaktert és a legtöbb felcserélést jelzi). */
function codeCheckChar_(body) {
  let s = 0;
  for (let i = 0; i < body.length; i++) {
    const v = CODE_ALPHABET.indexOf(body.charAt(i));
    if (v < 0) return '';
    s += (i + 1) * v;
  }
  return CODE_ALPHABET.charAt(s % CODE_ALPHABET.length);
}
function normalizeCode_(raw) { return String(raw == null ? '' : raw).toUpperCase().replace(/[^A-Z0-9]/g, ''); }
function codeIsWellFormed_(n) { return n.length === CODE_BODY_LEN + 1 && codeCheckChar_(n.slice(0, CODE_BODY_LEN)) === n.charAt(CODE_BODY_LEN); }
function formatCode_(n) { return n.slice(0, 4) + '-' + n.slice(4); }
function maskCode_(raw) {
  const n = normalizeCode_(raw);
  return n ? n.slice(0, 2) + '*'.repeat(Math.max(0, n.length - 2)) : '';
}
function newCode_() {
  const hex = Utilities.getUuid().replace(/-/g, '');
  let body = '';
  for (let i = 0; i < CODE_BODY_LEN; i++) body += CODE_ALPHABET.charAt(parseInt(hex.substr(i * 2, 2), 16) % CODE_ALPHABET.length);
  return body + codeCheckChar_(body);
}

/* ===================== BELÉPÉSI PONTOK ===================== */

function doPost(e) {
  let p = null, master = null, ss = null, idRow = null, isFirst = false;
  const lock = LockService.getScriptLock();

  try {
    p = parsePayload_(e);
    lock.waitLock(25000);
    try {
      master = getMaster_();

      // --- azonosítás: kód -> diák (név, osztály) ---
      let reject = '';
      let student = null;
      if (!codeIsWellFormed_(p.normCode)) reject = 'hibás kódformátum';
      else if (!(student = findStudent_(master, p.normCode))) reject = 'ismeretlen kód';
      else if (!student.active) reject = 'inaktív diák';
      else if (student.classId !== p.classId) reject = 'másik osztály kódja';
      else if (!student.id) throw new Error('A Diakok lapon hiányzik a(z) "' + student.name + '" azonosítója – futtasd a kodokGeneralasa()-t.');
      if (reject) {
        writeRow_(ensureSheet_(master, CONFIG.SHEET_REJECTED, REJECT_HEADERS),
          [new Date(), reject, maskCode_(p.codeRaw), p.classId, p.assignmentType, p.submissionId], [1], []);
        return json_({ ok: false, error: 'azonosítás' });
      }
      p.name = student.name; p.studentId = student.id;
      p.sheetName = studentSheetName_(student.name, p.classId);
      p.key = student.id + '|' + p.setId + '|' + p.assignmentType;

      ss = getClassSpreadsheet_(p.classId);
      const idSheet = ensureSheet_(ss, CONFIG.SHEET_IDLOG, ID_HEADERS);
      if (inColumn_(idSheet, 2, p.submissionId)) {
        log_(ss, 'DUPLIKÁLT', p.submissionId, 'Már feldolgozott beküldés, kihagyva.');
        return json_({ ok: true, duplicate: true });
      }
      isFirst = !inColumn_(idSheet, 1, p.key);               // első beküldés = érvényes

      appendStudentRow_(ss, p, isFirst);                     // a diák saját lapjára MINDIG
      idRow = writeRow_(idSheet,
        [p.key, p.submissionId, new Date(), p.studentId, p.name, p.setId, p.score,
         isFirst ? 'ÉLES' : 'GYAKORLÁS', isFirst ? 'függőben' : 'nincs (gyakorlás)'],
        [3], [7]);
      SpreadsheetApp.flush();
    } finally {
      lock.releaseLock();
    }

    if (isFirst) {                                            // e-mail csak az első, érvényes beküldésről
      let status = 'kikapcsolva';
      if (CONFIG.SEND_EMAIL) {
        try { sendEmail_(p); status = 'elküldve'; }
        catch (err) { status = 'HIBA'; log_(ss, 'E-MAIL HIBA', p.submissionId, String(err)); }
      }
      ss.getSheetByName(CONFIG.SHEET_IDLOG).getRange(idRow, 9).setValue(status);
    }
    return json_({ ok: true, first: isFirst });

  } catch (err) {
    let target = ss || master;
    if (!target) { try { target = getMaster_(); } catch (e2) { /* csak a Végrehajtások panelre naplózunk */ } }
    log_(target, 'HIBA', p ? p.submissionId : '', String(err && err.stack || err));
    return json_({ ok: false, error: 'szerverhiba' });
  }
}

/** GET: ?akcio=tartalom&osztaly=7 -> az osztály tartalomjegyzéke (JSON). Más kérésre csak életjel. */
function doGet(e) {
  try {
    const q = (e && e.parameter) || {};
    if (q.akcio === 'tartalom') return json_(getContent_(String(q.osztaly || '')));
  } catch (err) {
    log_(null, 'GET HIBA', '', String(err));
    return json_({ ok: false, error: 'szerverhiba' });
  }
  return ContentService.createTextOutput('A szolgáltatás működik.');
}

/* ===================== BEMENET ===================== */

function clean_(v, max) { return String(v == null ? '' : v).slice(0, max || CONFIG.MAX_TEXT); }
function cleanArr_(a) { return Array.isArray(a) ? a.slice(0, 20).map(function (x) { return clean_(x, 300); }) : []; }

function parsePayload_(e) {
  if (!e || !e.postData || !e.postData.contents) throw new Error('Üres kérés.');
  const p = JSON.parse(e.postData.contents);

  const submissionId = clean_(p.submissionId, 100);
  if (!submissionId) throw new Error('Hiányzik a submissionId.');
  const classId = clean_(p.classId, 20);
  if (!Object.prototype.hasOwnProperty.call(CLASSES, classId)) throw new Error('Ismeretlen vagy hiányzó classId: ' + classId);
  const assignmentType = clean_(p.assignmentType, 60);
  if (!/^[A-Za-z0-9_-]{1,60}$/.test(assignmentType)) throw new Error('Hiányzó vagy érvénytelen assignmentType.');
  const setId = clean_(p.setId != null ? p.setId : p.group, 40);
  if (!/^[A-Za-z0-9_.-]{1,40}$/.test(setId)) throw new Error('Hiányzó vagy érvénytelen setId.');

  const codeRaw = clean_(p.code, 30);
  const score = Number(p.score) || 0, total = Number(p.total) || 0;
  const rawTasks = (Array.isArray(p.tasks) ? p.tasks : [])
    .filter(function (t) { return t && typeof t === 'object'; }).slice(0, CONFIG.MAX_TASKS);

  return {
    submissionId: submissionId, sentAt: clean_(p.sentAt, 40),
    classId: classId, assignmentType: assignmentType, setId: setId,
    taskTitle: clean_(p.taskTitle, 60).replace(/[\r\n]+/g, ' ').trim() || 'Feladatlap',
    codeRaw: codeRaw, normCode: normalizeCode_(codeRaw),
    score: score, total: total,
    grade: clean_(p.grade, 100).replace(/[\r\n]+/g, ' ').trim() || (score + ' / ' + total),
    reason: REASONS[p.reason] ? p.reason : 'befejezve',
    tasks: rawTasks.map(function (t) {
      return {
        expression: clean_(t.expression, 200), correct: !!t.correct,
        finalAnswer: clean_(t.finalAnswer, 50), correctAnswer: clean_(t.correctAnswer, 50),
        seconds: Number(t.seconds) || 0, stepErrors: clean_(t.stepErrors, 300),
        studentLines: cleanArr_(t.studentLines), correctLines: cleanArr_(t.correctLines), feedback: cleanArr_(t.feedback),
      };
    }),
  };
}

/** "Teszt Elek" + "6a" -> "Teszt_Elek_6a" */
function studentSheetName_(name, classId) {
  const c = function (s) { return String(s).trim().replace(/\s+/g, '_').replace(/[^\p{L}\p{N}_-]/gu, ''); };
  let n = [c(name), c(classId)].filter(Boolean).join('_').slice(0, 90) || 'Ismeretlen';
  if (n === CONFIG.SHEET_IDLOG || n === CONFIG.SHEET_LOG) n = 'D_' + n;
  return n;
}

function findStudent_(master, normCode) {
  const sh = master.getSheetByName(CONFIG.SHEET_STUDENTS);
  if (!sh) return null;
  const v = sh.getDataRange().getValues();
  for (let i = 1; i < v.length; i++) {
    if (normalizeCode_(v[i][1]) === normCode) {
      return {
        id: String(v[i][0]).trim(), name: String(v[i][2]).trim(), classId: String(v[i][3]).trim(),
        active: !/^(nem|n|0|false|hamis)$/i.test(String(v[i][4]).trim()),
      };
    }
  }
  return null;
}

/* ===================== TARTALOMJEGYZÉK (GET) ===================== */

function getContent_(classId) {
  if (!Object.prototype.hasOwnProperty.call(CLASSES, classId)) return { ok: false, error: 'ismeretlen osztály' };
  const cache = CacheService.getScriptCache(), ckey = 'tartalom_' + classId;
  const hit = cache.get(ckey);
  if (hit) return JSON.parse(hit);

  const sh = getMaster_().getSheetByName(CONFIG.SHEET_CONTENT);
  const rows = sh ? sh.getDataRange().getValues().slice(1) : [];
  const tz = Session.getScriptTimeZone();
  const items = rows.filter(function (r) {
    const st = String(r[6]).trim() || 'aktív';
    return String(r[0]).trim() === classId && st !== 'rejtett' && String(r[3]).trim();
  }).map(function (r) {
    const d = r[1] instanceof Date ? Utilities.formatDate(r[1], tz, 'yyyy-MM-dd') : String(r[1]).trim();
    return { datum: d, tipus: String(r[2]).trim(), cim: String(r[3]).trim(), leiras: String(r[4]).trim(),
             link: String(r[5]).trim(), allapot: String(r[6]).trim() || 'aktív' };
  }).sort(function (a, b) { return a.datum < b.datum ? 1 : (a.datum > b.datum ? -1 : 0); });   // legújabb elöl

  const out = { ok: true, osztaly: classId, items: items };
  cache.put(ckey, JSON.stringify(out), CONFIG.CONTENT_CACHE_SECONDS);
  return out;
}

/* ===================== TÁBLÁZAT ===================== */

function appendStudentRow_(ss, p, isFirst) {
  const sh = ensureSheet_(ss, p.sheetName, SHEET_HEADERS);
  const details = p.tasks.map(function (t, i) {
    return (i + 1) + '. ' + (t.correct ? '✔' : '✘') + ' ' + t.expression + ' | válasz: ' + (t.finalAnswer || '—') +
           ' | helyes: ' + t.correctAnswer + ' | ' + t.seconds + ' mp' +
           (t.studentLines.length ? ' | lépések: ' + t.studentLines.join(' ⏎ ') : '') +
           (t.stepErrors ? ' | hibák: ' + t.stepErrors : '');
  }).join('\n').slice(0, 20000);
  writeRow_(sh, [new Date(), safeDate_(p.sentAt), p.assignmentType, p.taskTitle, p.setId, p.score, p.total, p.grade,
                 isFirst ? 'ÉLES' : 'GYAKORLÁS', REASONS[p.reason] + (isFirst ? '' : ' ' + CONFIG.PRACTICE_NOTE),
                 p.submissionId, details], [1, 2], [6, 7]);
}

/** A következő szabad sorba ír. Dátum- és számoszlopokon kívül minden sima szöveg ('@'), így a "=…" jellegű
    beírások nem értelmeződnek képletként. Visszaadja a sor számát. */
function writeRow_(sh, values, dateCols, numCols) {
  const r = sh.getLastRow() + 1;
  const fm = [values.map(function (_, i) {
    const c = i + 1;
    return dateCols.indexOf(c) >= 0 ? 'yyyy-mm-dd hh:mm:ss' : (numCols.indexOf(c) >= 0 ? '0' : '@');
  })];
  sh.getRange(r, 1, 1, values.length).setNumberFormats(fm).setValues([values]);
  return r;
}

/* ===================== DRIVE: MAPPA ÉS TÁBLÁZATOK ===================== */

function getRootFolder_() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('ROOT_FOLDER_ID');
  if (id) { try { const f = DriveApp.getFolderById(id); if (!f.isTrashed()) return f; } catch (e) { /* újrakeressük */ } }
  const it = DriveApp.getFoldersByName(CONFIG.ROOT_FOLDER);
  const folder = it.hasNext() ? it.next() : DriveApp.createFolder(CONFIG.ROOT_FOLDER);
  props.setProperty('ROOT_FOLDER_ID', folder.getId());
  return folder;
}

/** Név szerinti táblázat a gyökérmappában; ha nincs, létrehozza (az első lapot átnevezi és fejléccel látja el). */
function getSpreadsheetByName_(name, firstSheet, firstHeaders) {
  const props = PropertiesService.getScriptProperties(), key = 'SS_ID_' + name;
  const cached = props.getProperty(key);
  if (cached) { try { if (!DriveApp.getFileById(cached).isTrashed()) return SpreadsheetApp.openById(cached); } catch (e) { /* újrakeressük */ } }

  const root = getRootFolder_();
  const it = root.getFilesByName(name);
  while (it.hasNext()) {
    const f = it.next();
    if (f.getMimeType() === MimeType.GOOGLE_SHEETS) { props.setProperty(key, f.getId()); return SpreadsheetApp.openById(f.getId()); }
  }
  const ss = SpreadsheetApp.create(name);
  DriveApp.getFileById(ss.getId()).moveTo(root);
  const first = ss.getSheets()[0];
  first.setName(firstSheet);
  first.getRange(1, 1, 1, firstHeaders.length).setValues([firstHeaders]).setFontWeight('bold').setBackground('#e8eaf6');
  first.setFrozenRows(1);
  props.setProperty(key, ss.getId());
  return ss;
}
function getMaster_() { return getSpreadsheetByName_(CONFIG.MASTER_NAME, CONFIG.SHEET_STUDENTS, STUDENT_HEADERS); }
function getClassSpreadsheet_(classId) { return getSpreadsheetByName_('osztaly_' + classId, CONFIG.SHEET_IDLOG, ID_HEADERS); }

function ensureSheet_(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold').setBackground('#e8eaf6');
    sh.setFrozenRows(1);
  }
  return sh;
}

function inColumn_(sheet, col, value) {
  const last = sheet.getLastRow();
  if (last < 2) return false;
  return !!sheet.getRange(2, col, last - 1, 1).createTextFinder(value).matchEntireCell(true).findNext();
}

function log_(ss, type, id, message) {
  try {
    if (!ss) { Logger.log(type + ' | ' + id + ' | ' + message); return; }
    writeRow_(ensureSheet_(ss, CONFIG.SHEET_LOG, LOG_HEADERS),
      [new Date(), type, id || '', String(message).slice(0, 1500)], [1], []);
  } catch (e) { /* a naplózás hibája ne törjön tovább */ }
}

/* ===================== E-MAIL (csak az első, érvényes beküldésről) ===================== */

function sendEmail_(p) {
  MailApp.sendEmail({
    to: CONFIG.TEACHER_EMAIL,
    subject: '[' + p.taskTitle + ' – ' + CLASSES[p.classId] + ' – ' + p.assignmentType + '] ' + p.name + ' – ' +
             p.setId + '. feladatsor: ' + p.score + '/' + p.total,
    body: p.name + ' – ' + p.setId + '. feladatsor: ' + p.score + '/' + p.total + '. ' + p.grade +
          ' (a részletekhez HTML-képes levelezőt használj).',
    htmlBody: buildEmailHtml_(p),
    name: p.taskTitle,
  });
}

function esc_(v) {
  return String(v == null ? '' : v).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}

function buildEmailHtml_(p) {
  const mono = 'font-family:Consolas,Menlo,monospace;font-size:13px;';
  const pass = p.total > 0 && p.score === p.total;
  let html = '<div style="font-family:Arial,Helvetica,sans-serif;max-width:760px;margin:0 auto;color:#222">' +
    '<h2 style="margin:0 0 4px">' + esc_(p.taskTitle) + ' – ' + esc_(p.setId) + '. feladatsor</h2>' +
    '<p style="margin:0 0 10px;color:#555">' + esc_(p.name) + ' (' + esc_(CLASSES[p.classId]) + ')' +
    ' · beküldve: ' + esc_(formatDate_(p.sentAt)) + ' · típus: ' + esc_(p.assignmentType) + ' · lap: ' + esc_(p.sheetName) + '</p>' +
    '<div style="font-size:20px;font-weight:bold;margin-bottom:4px">Eredmény: ' + esc_(p.score) + ' / ' + esc_(p.total) + '</div>' +
    '<div style="margin-bottom:6px;font-weight:bold;color:' + (pass ? '#2e7d32' : '#b26a00') + '">Minősítés: ' + esc_(p.grade) + '</div>' +
    '<div style="margin-bottom:16px;color:#555;font-size:13px">Lezárás oka: ' + esc_(REASONS[p.reason]) +
    ' · megoldott feladat: ' + p.tasks.length + '</div>';

  p.tasks.forEach(function (t, i) {
    const color = t.correct ? '#2e7d32' : '#c62828', bg = t.correct ? '#e8f5e9' : '#fdecea';
    const rows = Math.max(t.studentLines.length, t.correctLines.length, 1);
    html += '<table role="presentation" cellspacing="0" cellpadding="0" width="100%" style="border-collapse:collapse;margin-bottom:18px;border:1px solid #ccc">' +
      '<tr><td colspan="2" style="background:' + bg + ';padding:8px 10px;border-bottom:1px solid #ccc"><b>' + (i + 1) + '. feladat</b> ' +
      '<span style="' + mono + '">' + esc_(t.expression) + '</span>' +
      '<span style="float:right;color:' + color + ';font-weight:bold">' + (t.correct ? '✔ helyes' : '✘ hibás') + '</span>' +
      '<div style="font-size:12px;color:#555;margin-top:3px">Beírt végeredmény: <b>' + esc_(t.finalAnswer || '—') +
      '</b> · helyes: <b>' + esc_(t.correctAnswer) + '</b> · idő: ' + esc_(t.seconds) + ' mp</div></td></tr>' +
      '<tr><th align="left" width="50%" style="padding:6px 10px;background:#f5f5f5;font-size:12px;border-right:1px solid #ddd">A diák megoldása</th>' +
      '<th align="left" width="50%" style="padding:6px 10px;background:#f5f5f5;font-size:12px">Helyes megoldásmenet</th></tr>';
    for (let k = 0; k < rows; k++) {
      const s = t.studentLines[k], c = t.correctLines[k];
      const none = (k === 0 && !t.studentLines.length) ? (t.correctLines.length ? '<i style="color:#888">nem írt lépéseket</i>' : '—') : '&nbsp;';
      html += '<tr><td valign="top" style="padding:4px 10px;border-top:1px solid #eee;border-right:1px solid #ddd;' + mono + '">' +
        (s ? esc_(s) : none) + '</td>' +
        '<td valign="top" style="padding:4px 10px;border-top:1px solid #eee;' + mono + '">' + (c ? esc_(c) : '&nbsp;') + '</td></tr>';
    }
    if (t.feedback.length) {
      html += '<tr><td colspan="2" style="padding:6px 10px;border-top:1px solid #ddd;background:#fffde7;font-size:12px"><b>Lépéselemzés:</b>' +
        '<ul style="margin:4px 0 0 18px;padding:0">' + t.feedback.map(function (m) { return '<li>' + esc_(m) + '</li>'; }).join('') + '</ul></td></tr>';
    }
    html += '</table>';
  });
  return html + '</div>';
}

function safeDate_(iso) { const d = new Date(iso); return isNaN(d.getTime()) ? new Date() : d; }
function formatDate_(iso) { return Utilities.formatDate(safeDate_(iso), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'); }

/** 1. Egyszer futtasd: létrehozza a mappát, a Torzs táblázatot és az osztálytáblázatokat; itt kéri el a jogosultságokat. */
function setup() {
  const master = getMaster_();
  ensureSheet_(master, CONFIG.SHEET_CONTENT, CONTENT_HEADERS);
  ensureSheet_(master, CONFIG.SHEET_REJECTED, REJECT_HEADERS);
  ensureSheet_(master, CONFIG.SHEET_LOG, LOG_HEADERS);

  const c = master.getSheetByName(CONFIG.SHEET_CONTENT);                 // legördülő listák a Tartalom lapon
  c.getRange(2, 3, 500, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(CONTENT_TYPES, true).setAllowInvalid(false).build());
  c.getRange(2, 7, 500, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(CONTENT_STATES, true).setAllowInvalid(false).build());
  c.getRange(2, 1, 500, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(CLASS_ORDER, true).setAllowInvalid(false).build());

  CLASS_ORDER.forEach(function (id) {
    const ss = getClassSpreadsheet_(id);
    ensureSheet_(ss, CONFIG.SHEET_LOG, LOG_HEADERS);
    Logger.log('osztaly_' + id + ' → ' + ss.getUrl());
  });
  Logger.log('Torzs → ' + master.getUrl());
  Logger.log('Hátralévő napi e-mail kvóta: ' + MailApp.getRemainingDailyQuota());
}

/** 2. A Diakok lapra illeszd be a névsort (Azonosító, Kód, Név, Osztály, Aktív), majd futtasd: a hiányzó azonosítókat
    és kódokat kitölti. A már meglévő kódokat nem bántja. Egy diák új kódjához töröld a Kód cellát, és futtasd újra. */
function kodokGeneralasa() {
  const sh = getMaster_().getSheetByName(CONFIG.SHEET_STUDENTS);
  const last = sh.getLastRow();
  if (last < 2) { Logger.log('A Diakok lap üres.'); return; }
  const rng = sh.getRange(2, 1, last - 1, STUDENT_HEADERS.length);
  const v = rng.getValues();
  const used = {}, count = {};
  v.forEach(function (r) { const n = normalizeCode_(r[1]); if (n) used[n] = true; });
  let madeCodes = 0, madeIds = 0, skipped = 0;

  v.forEach(function (r) {
    const name = String(r[2]).trim(), cls = String(r[3]).trim();
    if (!name) return;
    if (!Object.prototype.hasOwnProperty.call(CLASSES, cls)) { skipped++; Logger.log('Ismeretlen osztály, kihagyva: ' + name + ' / ' + cls); return; }
    if (!String(r[0]).trim()) {
      let n = 1;
      const re = new RegExp('^' + cls + '-(\\d+)$');
      v.forEach(function (x) { const m = re.exec(String(x[0]).trim()); if (m) n = Math.max(n, Number(m[1]) + 1); });
      r[0] = cls + '-' + (n < 10 ? '0' : '') + n; madeIds++;
    }
    if (!normalizeCode_(r[1])) {
      let c;
      do { c = newCode_(); } while (used[c]);
      used[c] = true; r[1] = formatCode_(c); madeCodes++;
    }
    if (!String(r[4]).trim()) r[4] = 'igen';
  });
  rng.setNumberFormat('@').setValues(v);
  Logger.log('Új azonosító: ' + madeIds + ' · új kód: ' + madeCodes + ' · kihagyva: ' + skipped);
}

/** 3. Nyomtatható kódlista (osztályonként, névsorban) a Torzs „Kodlap” lapján. Kinyomtatás után futtasd a kodlapTorlese()-t. */
function kodlapKeszitese() {
  const master = getMaster_();
  const v = master.getSheetByName(CONFIG.SHEET_STUDENTS).getDataRange().getValues().slice(1)
    .filter(function (r) { return String(r[2]).trim() && String(r[1]).trim(); });
  const order = CLASS_ORDER;
  v.sort(function (a, b) {
    const d = order.indexOf(String(a[3]).trim()) - order.indexOf(String(b[3]).trim());
    return d !== 0 ? d : String(a[2]).localeCompare(String(b[2]), 'hu');
  });
  let sh = master.getSheetByName('Kodlap');
  if (sh) master.deleteSheet(sh);
  sh = master.insertSheet('Kodlap');
  const out = [['Osztály', 'Név', 'Kód']];
  let prev = null;
  v.forEach(function (r) {
    const cls = String(r[3]).trim();
    if (prev !== null && cls !== prev) out.push(['', '', '']);
    out.push([CLASSES[cls] || cls, String(r[2]).trim(), String(r[1]).trim()]);
    prev = cls;
  });
  sh.getRange(1, 1, out.length, 3).setNumberFormat('@').setValues(out);
  sh.getRange(1, 1, 1, 3).setFontWeight('bold');
  sh.getRange(2, 3, out.length - 1, 1).setFontFamily('Consolas').setFontSize(13);
  sh.autoResizeColumns(1, 3);
  Logger.log('Kodlap kész: ' + (out.length - 1) + ' sor. Nyomtatás után futtasd a kodlapTorlese()-t.');
}
function kodlapTorlese() {
  const master = getMaster_(), sh = master.getSheetByName('Kodlap');
  if (sh) master.deleteSheet(sh);
}

/** Kipróbálás: az első kóddal rendelkező diák nevében küld 1. érvényes, 2. gyakorló, 3. duplikált és 4. hibás kódú beküldést.
    A teszt a diák saját lapjára és az Id_Log-ba ír, e-mail is megy: utána töröld a teszt lapot és sorokat. */
function tesztBekuldes() {
  const v = getMaster_().getSheetByName(CONFIG.SHEET_STUDENTS).getDataRange().getValues().slice(1).filter(function (r) { return r[1] && r[2]; });
  if (!v.length) { Logger.log('Előbb futtasd a kodokGeneralasa()-t.'); return; }
  const code = String(v[0][1]), cls = String(v[0][3]);
  const mk = function (id, score, c) {
    return { postData: { contents: JSON.stringify({
      submissionId: id, sentAt: new Date().toISOString(), reason: 'befejezve', code: c,
      classId: cls, assignmentType: 'Teszt', taskTitle: 'Teszt feladatlap', setId: '1', score: score, total: 2, grade: score === 2 ? 'Jeles (5)' : 'Nem megfelelt',
      tasks: [{ expression: '2 + 3 · 4', correct: true, finalAnswer: '14', correctAnswer: '14', seconds: 20,
                studentLines: ['3 · 4 = 12 → 2 + 12'], correctLines: ['3 · 4 = 12 → 2 + 12 = 14'], feedback: [] },
              { expression: '10 - 2 - 3', correct: score === 2, finalAnswer: score === 2 ? '5' : '9', correctAnswer: '5', seconds: 15,
                studentLines: [], correctLines: ['10 - 2 = 8 → 8 - 3 = 5'], feedback: [] }] }) } };
  };
  const base = 'teszt-' + Date.now();
  Logger.log('1. érvényes:   ' + doPost(mk(base + 'a', 2, code)).getContent());
  Logger.log('2. gyakorlás:  ' + doPost(mk(base + 'b', 1, code)).getContent());
  Logger.log('3. duplikált:  ' + doPost(mk(base + 'b', 1, code)).getContent());
  Logger.log('4. hibás kód:  ' + doPost(mk(base + 'c', 2, 'AAAA-AAAA')).getContent());
}
